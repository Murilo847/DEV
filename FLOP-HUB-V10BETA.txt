-- SMART BOUNDS
local function GetSmartPosition(currentPos, width, height)
    if not SmartPositionEnabled then return currentPos end
    local vp = Camera.ViewportSize
    local padding = 5 
    local halfWidth = width / 2
    local halfHeight = height / 2
    
    local absCenterX = (currentPos.X.Scale * vp.X) + currentPos.X.Offset
    local absCenterY = (currentPos.Y.Scale * vp.Y) + currentPos.Y.Offset
    
    local safeCenterX = math_clamp(absCenterX, halfWidth + padding, vp.X - halfWidth - padding)
    local safeCenterY = math_clamp(absCenterY, halfHeight + padding, vp.Y - halfHeight - padding)
    
    return UDim2_new(0, safeCenterX, 0, safeCenterY)
end

-- DRAG SYSTEM
local function MakeDraggable(trigger, target, isMainHub)
    local dragging, dragInput, dragStart, startPos
    
    table_insert(_G.SkyL_Connections, trigger.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = target.Position
            LastInteraction = tick()
            
            local con; con = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then 
                    dragging = false 
                    con:Disconnect()
                    
                    if isMainHub then
                        UserSavedPosition = target.Position
                        local finalPos = GetSmartPosition(target.Position, target.AbsoluteSize.X, target.AbsoluteSize.Y)
                        if finalPos ~= target.Position then
                            TweenService:Create(target, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = finalPos}):Play()
                            UserSavedPosition = finalPos
                        end
                    end
                end
            end)
        end
    end))
    
    table_insert(_G.SkyL_Connections, trigger.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))
    
    table_insert(_G.SkyL_Connections, UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            local newPos = UDim2_new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            target.Position = newPos
            LastInteraction = tick()
        end
    end))
end

    local ColorCache = {}
    local function GetCachedColorSequence(coresArray)
        -- Gera chave única convertendo Color3 para string
        local keyParts = {}
        for i=1, #coresArray do keyParts[i] = tostring(coresArray[i]) end
        local key = table.concat(keyParts, "_")
        
        if ColorCache[key] then return ColorCache[key] end

        local keypoints = {}
        if #coresArray == 1 then
             local c1 = coresArray[1]
             keypoints = {
                ColorSequenceKeypoint.new(0, c1),
                ColorSequenceKeypoint.new(1, c1)
            }
        else
            local maxCores = math.min(#coresArray, 20)
            local step = 1 / (maxCores - 1)
            for i = 1, maxCores do
                local time = (i - 1) * step
                if i == maxCores then time = 1 end
                table.insert(keypoints, ColorSequenceKeypoint.new(time, coresArray[i]))
            end
        end
        
        local seq = ColorSequence.new(keypoints)
        ColorCache[key] = seq
        return seq
    end

                    end
                    
                    if #coresEncontradas > 0 then
                        for _, p in pairs(Players:GetPlayers()) do
                            local pIdent = string.lower(p.Name)
                            local pDisp = string.lower(p.DisplayName)
                            if pIdent == usuario or pDisp == usuario then
                                NovasCoresAtivas[p.UserId] = coresEncontradas
                                -- Pinta imediatamente apenas se mudou ou se é a primeira vez
                                if not NEXUS_CoresAtivas[p.UserId] or #NEXUS_CoresAtivas[p.UserId] ~= #coresEncontradas then
                                     if p.Character then for _, obj in pairs(p.Character:GetDescendants()) do Nexus_Pintar(obj, coresEncontradas) end end
                                end
                            end
                        end
                    end
                end
            end
            NEXUS_CoresAtivas = NovasCoresAtivas
        end
    end

    -- Monitor de Projéteis e Magias (NEXUS)
    -- [OTIMIZAÇÃO] Monitor de Projéteis com Fila (Batch Queue)
    -- Corrige "Agatha Blast" (objetos movidos após spawn) s/ delay de alocação de thread
    local ObjectQueue = {}
    local NEXUS_QUEUE_MAX = 50
    local NEXUS_BATCH_SIZE = 10
    local Nexus_MonitorConnection = Workspace.DescendantAdded:Connect(function(obj)
        if #ObjectQueue < NEXUS_QUEUE_MAX and (obj:IsA("BasePart") or obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail") or obj:IsA("Light") or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("SelectionBox") or obj:IsA("SpecialMesh")) then
            ObjectQueue[#ObjectQueue + 1] = obj
        end
    end)
    
    local Nexus_QueueProcessor = RunService.Heartbeat:Connect(function()
        if #ObjectQueue == 0 or not NexusColorEnabled then if not NexusColorEnabled then table.clear(ObjectQueue) end return end
        
        -- 1. Cache de Alvos Ativos (Pre-calculation)
        -- Evita buscar players/characters/rootparts a cada objeto da fila
        local ActiveTargets = {} 
        for userId, coresArray in pairs(NEXUS_CoresAtivas) do
            local player = Players:GetPlayerByUserId(userId)
            if player and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                     -- Salva posição e cores para uso rápido
                    table.insert(ActiveTargets, {
                        Pos = hrp.Position,
                        Colors = coresArray
                    })
                end
            end
        end

        -- Se não tem ninguém com cor ativa, limpa a fila e retorna
        if #ActiveTargets == 0 then 
            table.clear(ObjectQueue)
            return 
        end

        -- 2. Processa fila atual (Batch limitado a NEXUS_BATCH_SIZE por frame)
        local processCount = math.min(#ObjectQueue, NEXUS_BATCH_SIZE)
        for idx = 1, processCount do
            local obj = ObjectQueue[idx]
            if obj and obj.Parent then
                local objPos = nil
                if obj:IsA("BasePart") then 
                    objPos = obj.Position 
                elseif obj.Parent then
                    if obj.Parent:IsA("BasePart") then 
                        objPos = obj.Parent.Position
                    elseif obj.Parent:IsA("Attachment") then 
                        objPos = obj.Parent.WorldPosition 
                    end
                end
                if objPos then
                    for _, target in ipairs(ActiveTargets) do
                        if (target.Pos - objPos).Magnitude < 25 then 
                            Nexus_Pintar(obj, target.Colors)
                            break
                        end
                    end
                end
            end
        end
        -- Remove itens processados, mantendo o resto na fila
        if processCount >= #ObjectQueue then
            table.clear(ObjectQueue)
        else
            for i = 1, processCount do table.remove(ObjectQueue, 1) end
        end
    end)
    table_insert(_G.SkyL_Connections, Nexus_QueueProcessor)
    table_insert(_G.SkyL_Connections, Nexus_MonitorConnection)

    -- Loop de Atualização Gist
    task.spawn(function()
        while _G.SkyL_Connections do
            if NexusColorEnabled then
                Nexus_LerBancoDeDados()
            end
            task.wait(NEXUS_TEMPO_ATUALIZACAO)
        end
    end)



    -- Evento CharAdded para repintar
    table_insert(_G.SkyL_Connections, LocalPlayer.CharacterAdded:Connect(function(c)
        task.wait(1)
        if NEXUS_CoresAtivas[LocalPlayer.UserId] then
            for _, obj in pairs(c:GetDescendants()) do Nexus_Pintar(obj, NEXUS_CoresAtivas[LocalPlayer.UserId]) end
        end
    end))
end

function Farm.ValidateTarget(target)
    if not target or not target.Parent then return false end
    local hum = target:FindFirstChild("Humanoid"); local hrp = target:FindFirstChild("HumanoidRootPart")
    if not hum or hum.Health <= 0 or not hrp then return false end
    if Players:GetPlayerFromCharacter(target) then return false end
    if hrp.Anchored == true then return false end 
    return true
end

function Farm.GetTarget()
    if Farm.CurrentTarget and Farm.ValidateTarget(Farm.CurrentTarget) and not table.find(Blacklist, Farm.CurrentTarget) then return Farm.CurrentTarget end
    local closest, dist = nil, 999999; local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    for _, v in pairs(Workspace:GetChildren()) do
        if v:IsA("Model") and v ~= LocalPlayer.Character and not table.find(Blacklist, v) and Farm.ValidateTarget(v) then
            local hrp = v:FindFirstChild("HumanoidRootPart")
            if hrp then local d = (hrp.Position - myRoot.Position).Magnitude; if d < dist then dist = d; closest = v end end
        end
    end
    Farm.CurrentTarget = closest; return closest
end

-- Chest Logic
function Farm.IsChestIgnored(name)
    local lowerName = name:lower()
    for _, ignore in ipairs(Farm.ChestIgnore) do if lowerName:find(ignore) then return true end end
    return false
end

local ChestCache = {}
local ChestCacheMap = {} -- lookup table
local InitialChestScanDone = false

local function AddChestToCache(instance)
    if not instance or not instance.Parent then return end
    if ChestCacheMap[instance] then return end
    
    local prompt = instance:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not prompt then return end
    
    local name = instance.Name:lower()
    local promptText = prompt.ObjectText:lower()
    
    if not Farm.IsChestIgnored(instance.Name) and not (instance.Parent and Farm.IsChestIgnored(instance.Parent.Name)) then
        if name:find("chest") or name:find("baú") or name:find("bau") or name:find("crate") or promptText:find("chest") or promptText:find("baú") or promptText:find("bau") or promptText:find("crate") then
            local chestData = {Prompt = prompt, Instance = instance}
            table.insert(ChestCache, chestData)
            ChestCacheMap[instance] = chestData
        end
    end
end

local function RemoveChestFromCache(instance)
    if ChestCacheMap[instance] then
        local data = ChestCacheMap[instance]
        local idx = table.find(ChestCache, data)
        if idx then table.remove(ChestCache, idx) end
        ChestCacheMap[instance] = nil
    end
end

-- Initialize the Cache
local function InitChestCache()
    if InitialChestScanDone then return end
    InitialChestScanDone = true
    
    -- Throttled initial scan to prevent locking
    task.spawn(function()
        local descendants = Workspace:GetDescendants()
        for i, desc in ipairs(descendants) do
            if desc:IsA("ProximityPrompt") and desc.Parent then
                local instance = desc.Parent
                if instance:IsA("Attachment") then instance = instance.Parent end
                if instance and (instance:IsA("BasePart") or instance:IsA("Model")) then
                    AddChestToCache(instance)
                end
            end
            if i % 1000 == 0 then task.wait() end -- Yield every 1000 items
        end
    end)
    
    -- Event listeners for new/removed chests
    table.insert(_G.SkyL_Connections, Workspace.DescendantAdded:Connect(function(desc)
        if desc:IsA("ProximityPrompt") then
            task.wait(0.1) -- wait for parent hierarchy to settle
            if not desc.Parent then return end
            local instance = desc.Parent
            if instance:IsA("Attachment") then instance = instance.Parent end
            if instance and (instance:IsA("BasePart") or instance:IsA("Model")) then
                AddChestToCache(instance)
            end
        end
    end))
    
    table.insert(_G.SkyL_Connections, Workspace.DescendantRemoving:Connect(function(desc)
        if desc:IsA("ProximityPrompt") then
            local instance = desc.Parent
            if instance:IsA("Attachment") then instance = instance.Parent end
            if instance then RemoveChestFromCache(instance) end
        end
    end))
end

function Farm.GetValidChests()
    if not InitialChestScanDone then InitChestCache() end
    
    local valid = {}
    for i = #ChestCache, 1, -1 do
        local c = ChestCache[i]
        if c.Prompt and c.Prompt.Parent and c.Instance and c.Instance.Parent then
            valid[#valid + 1] = c
        else
            -- Cleanup invalid cache entries
            table.remove(ChestCache, i)
            if c.Instance then ChestCacheMap[c.Instance] = nil end
        end
    end
    return valid
end

-- Helper: coleta um unico bau com smooth teleport
function Farm.SmoothCollectOne(chest)
    if not chest.Instance or not chest.Instance.Parent then return false end
    if not chest.Prompt or not chest.Prompt.Parent then return false end

    local inst = chest.Instance
    local chestPart = nil
    if inst:IsA("Model") then
        chestPart = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
    elseif inst:IsA("BasePart") then
        chestPart = inst
    end
    if not chestPart then return false end

    local pos = chestPart.Position
    local savedChestCF = chestPart.CFrame

    -- Heartbeat teleporta pro bau (server-side)
    Farm.SmoothTargetPos = pos + Vector3_new(0, 2, 0)
    Farm.SmoothModeActive = true

    -- Gruda bau no jogador (client-side)
    local followConn = RunService.RenderStepped:Connect(function()
        if Farm.PlayerRealCFrame and chestPart.Parent then
            if Farm.NPCEnabled then
                chestPart.CFrame = Farm.PlayerRealCFrame * CFrame.new(0, -3, 0)
            else
                chestPart.CFrame = Farm.PlayerRealCFrame * CFrame.new(0, 0, -4)
            end
        end
    end)

    -- Configura prompt
    local origDist = chest.Prompt.MaxActivationDistance
    local origHold = chest.Prompt.HoldDuration
    local origLOS = chest.Prompt.RequiresLineOfSight
    chest.Prompt.MaxActivationDistance = 9999
    chest.Prompt.HoldDuration = 0
    chest.Prompt.RequiresLineOfSight = false

    task.wait(0.15)

    for k = 1, Farm.SMOOTH_SPAM_COUNT do
        pcall(function() fireproximityprompt(chest.Prompt, 0, true) end)
        if k < Farm.SMOOTH_SPAM_COUNT then task.wait(Farm.SMOOTH_SPAM_DELAY) end
    end

    pcall(function()
        chest.Prompt.MaxActivationDistance = origDist
        chest.Prompt.HoldDuration = origHold
        chest.Prompt.RequiresLineOfSight = origLOS
    end)

    pcall(function() followConn:Disconnect() end)
    if savedChestCF and chestPart.Parent then
        pcall(function() chestPart.CFrame = savedChestCF end)
    end

    Farm.SmoothModeActive = false
    Farm.SmoothTargetPos = nil
    return true
end

local Nexus = {}

function Nexus.DeepFind(exactName)
    local map = Workspace:FindFirstChild("Map")
    local interactables = Workspace:FindFirstChild("Interactables")

    if exactName == "Darkhold" then
        if map and map:FindFirstChild("Darkhold") then
            local visual = map.Darkhold:FindFirstChild("Visual")
            if visual then return visual end
            return map.Darkhold
        end
    elseif exactName == "TimeStone" then
        if map and map:FindFirstChild("Sanctum") then
            local timeStone = map.Sanctum:FindFirstChild("TimeStone")
            if timeStone then return timeStone end
        end
    elseif exactName == "SpaceStone" then
        if interactables and interactables:FindFirstChild("SpaceStone") then
            return interactables.SpaceStone
        end
        local spaceStone = Workspace:FindFirstChild("SpaceStone")
        if spaceStone then return spaceStone end
    end

    local folders = {
        Workspace:FindFirstChild("Sanctum"),
        Workspace:FindFirstChild("Interactables"),
        Workspace
    }
    for _, folder in pairs(folders) do
        if folder then
            local item = folder:FindFirstChild(exactName)
            if item then return item end
        end
    end
    return nil
end

function Nexus.Collect(exactName, instantMode)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild(STR_HRP) then return end

    local targetModel = Nexus.DeepFind(exactName)
    if not targetModel then
        AddLog(exactName .. " nao encontrado", ColorRed)
        return
    end

    local dest = targetModel:FindFirstChild("Handle")
        or targetModel:FindFirstChild("Root")
        or targetModel:FindFirstChild("HumanoidRootPart")
        or (targetModel:IsA("BasePart") and targetModel)
        or targetModel:FindFirstChildWhichIsA("BasePart")

    local prompt = targetModel:FindFirstChildWhichIsA("ProximityPrompt", true)

    if not dest or not prompt then
        AddLog("Sem prompt: " .. exactName, ColorRed)
        return
    end

    AddLog("Coletando " .. exactName .. "...", Color3.fromRGB(255, 255, 0))

    -- Heartbeat teleporta pro item (server-side)
    Farm.SmoothTargetPos = dest.CFrame.Position + Vector3_new(0, 2, 0)
    Farm.SmoothModeActive = true

    -- Gruda item no jogador (client-side)
    local savedCFrame = dest.CFrame
    local followConn = RunService.RenderStepped:Connect(function()
        if Farm.PlayerRealCFrame and dest.Parent then
            dest.CFrame = Farm.PlayerRealCFrame * CFrame.new(0, 0, -4)
        end
    end)

    -- Configura prompt
    local origDist = prompt.MaxActivationDistance
    local origHold = prompt.HoldDuration
    local origLOS = prompt.RequiresLineOfSight
    prompt.MaxActivationDistance = 9999
    prompt.RequiresLineOfSight = false
    if instantMode then prompt.HoldDuration = 0 end

    task_wait(0.2)

    for k = 1, Farm.SMOOTH_SPAM_COUNT do
        pcall(function() fireproximityprompt(prompt, 0, true) end)
        if k < Farm.SMOOTH_SPAM_COUNT then task_wait(Farm.SMOOTH_SPAM_DELAY) end
    end

    pcall(function()
        prompt.MaxActivationDistance = origDist
        prompt.HoldDuration = origHold
        prompt.RequiresLineOfSight = origLOS
    end)

    pcall(function() followConn:Disconnect() end)
    if savedCFrame and dest.Parent then
        pcall(function() dest.CFrame = savedCFrame end)
    end

    Farm.SmoothModeActive = false
    Farm.SmoothTargetPos = nil

    pcall(function()
        local humC = char:FindFirstChild(STR_HUM)
        if humC then humC.CameraOffset = Vector3_zero end
    end)

    AddLog(exactName .. " coletado!", ColorGreen)
end

function Farm.GetActiveArtifactToCatch(myOwned)
    -- Procura no workspace objetos caídos que queremos (prioridade: Time, Power, Dark)
    myOwned = myOwned or {}
    for _, artName in ipairs(ArtifactsPriorityList) do
        if Farm.AutoCatchArtifacts[artName] and not myOwned[artName] then
            local art = Nexus.DeepFind(artName)
            -- Se achou no Workspace e não está num jogador (geralmente não tem Humanoid pai)
            if art and not art.Parent:FindFirstChild(STR_HUM) then
                return artName, art
            end
        end
    end
    return nil, nil
end

function Farm.CollectChests()
    AddLog("COLETANDO BAÚS...", Color3.fromRGB(255, 255, 0))
    local chests = Farm.GetValidChests()
    if #chests == 0 then AddLog("NENHUM BAÚ ENCONTRADO", Color3.fromRGB(255, 0, 0)); return end

    for i, chest in ipairs(chests) do
        if not Farm.ChestEnabled then break end
        Farm.SmoothCollectOne(chest)
    end
    AddLog("COLETA FINALIZADA", Color3.fromRGB(0, 255, 0))
end

end

local SkyL_FlyBV = nil
local SkyL_FlyBG = nil
local WasFlying = false

local function UpdateMovement()
    if not LocalPlayer.Character then return end
    local hum = LocalPlayer.Character:FindFirstChild(STR_HUM); local hrp = LocalPlayer.Character:FindFirstChild(STR_HRP)
    if not hrp then return end

    -- SEMPRE salva posicao REAL antes de qualquer coisa
    Farm.PlayerRealCFrame = hrp.CFrame

    if hum and not FlyMode then 
        SetProp(hum, "WalkSpeed", 16 + (CustomSpeed * 10))
        if CustomJump > 0 then SetProp(hum, "JumpPower", CustomJump) end 
    end
    
    if FlyMode and hrp and hum then
        if not WasFlying then
            WasFlying = true
            SetProp(hum, "PlatformStand", true)
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.Swimming) end)
            
            if SkyL_FlyBV then SkyL_FlyBV:Destroy() end
            SkyL_FlyBV = Instance.new("BodyVelocity")
            SkyL_FlyBV.Name = "SkyL_FlyBV"
            SkyL_FlyBV.MaxForce = Vector3_new(9e9, 9e9, 9e9)
            SkyL_FlyBV.Velocity = Vector3_new(0, 0.1, 0)
            SkyL_FlyBV.Parent = hrp
            
            if SkyL_FlyBG then SkyL_FlyBG:Destroy() end
            SkyL_FlyBG = Instance.new("BodyGyro")
            SkyL_FlyBG.Name = "SkyL_FlyBG"
            SkyL_FlyBG.P = 9e4
            SkyL_FlyBG.MaxTorque = Vector3_new(9e9, 9e9, 9e9)
            SkyL_FlyBG.CFrame = hrp.CFrame
            SkyL_FlyBG.Parent = hrp
            
            local animate = LocalPlayer.Character:FindFirstChild("Animate")
            if animate then animate.Disabled = true end
            for _, v in pairs(hum:GetPlayingAnimationTracks()) do v:AdjustSpeed(0) end
        end
        
        local cam = workspace.CurrentCamera
        local vMove = Vector3_zero
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vMove = vMove + Vector3_new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vMove = vMove - Vector3_new(0, 1, 0) end
        
        local moveDir = hum.MoveDirection
        local move3D = vMove
        if moveDir.Magnitude > 0 then
            local _, yRot, _ = cam.CFrame:ToEulerAnglesYXZ()
            local camFlat = CFrame.Angles(0, yRot, 0)
            local rawInput = camFlat:VectorToObjectSpace(moveDir)
            move3D = move3D + (cam.CFrame.RightVector * rawInput.X) + (cam.CFrame.LookVector * -rawInput.Z)
        end
        if move3D.Magnitude > 0 then move3D = move3D.Unit end
        
        local speedMultiplier = (CustomSpeed == 0 and 1) or CustomSpeed
        
        if move3D.Magnitude > 0 then
            LocalPlayer.Character:TranslateBy(move3D * speedMultiplier)
        end
        
        if SkyL_FlyBG then SkyL_FlyBG.CFrame = cam.CFrame end
        Farm.PlayerRealCFrame = hrp.CFrame
    else
        if WasFlying then
            WasFlying = false
            if hum and not Farm.NPCEnabled then SetProp(hum, "PlatformStand", false) end
            if SkyL_FlyBV then SkyL_FlyBV:Destroy(); SkyL_FlyBV = nil end
            if SkyL_FlyBG then SkyL_FlyBG:Destroy(); SkyL_FlyBG = nil end
            local animate = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Animate")
            if animate then animate.Disabled = false end
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics) end)
        end
    end
    
    if NoclipMode then for _, part in ipairs(CachedCharParts) do if part.Parent and part.CanCollide then part.CanCollide = false end end end

    -- SMOOTH TELEPORT & INVISIBLE LOGIC: Executa DEPOIS do movement
    -- Garante que NUNCA vai haver conflitos de yield no Heartbeat!
    if Farm.SmoothModeActive and Farm.SmoothTargetPos and hum then
        local currentCFrame = hrp.CFrame
        local offset = Farm.SmoothTargetPos - currentCFrame.Position
        local destinoCFrame = currentCFrame + offset

        hrp.CFrame = destinoCFrame
        hum.CameraOffset = destinoCFrame:ToObjectSpace(CFrame.new(currentCFrame.Position)).Position
        RunService.RenderStepped:Wait()
        hrp.CFrame = currentCFrame
        hum.CameraOffset = Vector3_zero
    elseif InvisibleMode and hum and hrp then
        -- Lógica de invisibilidade (The Void Logic)
        -- Processa TP fantasma
        if InvisibleDesiredCFrame then
            hrp.CFrame = InvisibleDesiredCFrame
            InvisibleDesiredCFrame = nil
            Farm.PlayerRealCFrame = hrp.CFrame -- Atualiza o RealCFrame para ser a âncora certa
        end
        
        local currentCFrame = hrp.CFrame
        local originalCamOffset = hum.CameraOffset
        
        -- Teleport to Void
        local voidCFrame = currentCFrame + Vector3_new(0, VOID_DEPTH, 0)
        hrp.CFrame = voidCFrame
        
        -- Compensate Camera
        hum.CameraOffset = voidCFrame:ToObjectSpace(CFrame.new(currentCFrame.Position)).Position
        
        -- Wait for render (server sees void)
        RunService.RenderStepped:Wait()
        
        -- Restore position
        hrp.CFrame = currentCFrame
        hum.CameraOffset = originalCamOffset
    else
        -- Safety catch para reverter a câmera caso invisível seja desligado
        if hum and hum.CameraOffset.Magnitude > 100 then
            hum.CameraOffset = Vector3_zero
        end
    end
end

local function ValidateTarget(target)
    if not target or not target.Parent then return false end
    local hum = target:FindFirstChild(STR_HUM); 
    if not hum or hum.Health <= 0 then return false end
    local hrp = target:FindFirstChild(STR_HRP)
    if not hrp then return false end
    if Players:GetPlayerFromCharacter(target) then return false end
    if target.Name == "SkyL_SafeSpot" then return false end; if hrp.Anchored == true then return false end 
    return true
end

-- Otimização: Não alocar tabela em cada frame
local function GetTarget()
    local closest, dist = nil, 999999; local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(STR_HRP)
    if not myRoot then return nil end
    local children = Workspace:GetChildren() 
    local myPos = myRoot.Position
    for i = 1, #children do
        local v = children[i]
        if v:IsA("Model") and v ~= LocalPlayer.Character and not table_find(Blacklist, v) and ValidateTarget(v) then
            local hrp = v:FindFirstChild(STR_HRP); 
            if hrp then 
                local d = (hrp.Position - myPos).Magnitude; 
                if d < dist then dist = d; closest = v end 
            end
        end
    end
    return closest
end

-- // MANAGE ESP V2 // --
local function ManageESP(v, col, line, name)
    if not v.Character then return end
    local hrp = v.Character:FindFirstChild(STR_HRP); local head = v.Character:FindFirstChild(STR_HEAD); local hum = v.Character:FindFirstChild(STR_HUM)
    if not hrp or not head then return end
    
    -- Distance Check
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(STR_HRP) then 
        if (hrp.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude > MaxRenderDistance then 
            ClearVisuals(v); return 
        end 
    end
    
    -- Line Logic (Recycle)
    local beam = hrp:FindFirstChild("ArtifactBeam")
    if line then
        local myAtt = GetLocalAttachment()
        if not beam then 
            local att = hrp:FindFirstChild("EspAttTarget") or Instance.new("Attachment", hrp); att.Name="EspAttTarget"
            beam = Instance.new("Beam", hrp); beam.Name="ArtifactBeam"; beam.Attachment1=att; beam.FaceCamera=true; beam.Width0=0.1; beam.Width1=0.1 
        end
        if beam and myAtt then 
            SetProp(beam, "Attachment0", myAtt)
            beam.Color = ColorSequence_new(col)
            SetProp(beam, "Enabled", true)
        end
    else 
        if beam then SetProp(beam, "Enabled", false) end 
    end
    
    -- Name Logic (Recycle)
    local tag = head:FindFirstChild("ArtifactNameTag")
    if name then
        local l
        if not tag then 
            tag = Instance.new("BillboardGui", head); tag.Name="ArtifactNameTag"; tag.AlwaysOnTop=true; tag.Size=UDim2_new(0,200,0,50); tag.StudsOffset=Vector3_new(0,3,0)
            l = Instance.new("TextLabel", tag); l.Name="L"; l.Size=UDim2_new(1,0,1,0); l.BackgroundTransparency=1; l.Font=Enum.Font.Nunito; l.TextSize=16; l.TextStrokeTransparency=0; l.TextStrokeColor3=ColorBlack
        else
            l = tag:FindFirstChild("L")
            SetProp(tag, "Enabled", true)
        end
        
        if l then 
            local hp = hum and math_floor(hum.Health) or 0; local max = hum and math_floor(hum.MaxHealth) or 100
            local newText = v.DisplayName.." ["..hp.."/"..max.."]"
            SetProp(l, "Text", newText)
            SetProp(l, "TextColor3", col)
        end
    else 
        if tag then SetProp(tag, "Enabled", false) end 
    end
end

local function getClosestPlayerToMouse()
    local closest, shortest = nil, math.huge; local mouse = UserInputService:GetMouseLocation(); local players = Players:GetPlayers()
    for i=1, #players do
        local v = players[i]
        local vName = v.Name
        if v ~= LocalPlayer and v.Character and not Whitelist[vName] then
            local hrp = v.Character:FindFirstChild(STR_HRP); local hum = v.Character:FindFirstChild(STR_HUM)
            if hrp and hum and hum.Health > 0 then 
                local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position); 
                if onScreen then 
                    local dist = (Vector2_new(pos.X, pos.Y) - mouse).Magnitude; 
                    if dist < shortest then shortest = dist; closest = v end 
                end 
            end
        end
    end
    return closest
end

-- [NEW] SMART TARGET LOGIC (Sticky Focus)
-- Prioritizes: 1. Screen Distance (Mouse), 2. Depth (Camera Z)
-- Prevents clicking through players or hitting background targets
local function GetSmartTarget(checkPos, radius)
    local mouseLoc = checkPos or UserInputService:GetMouseLocation()
    local bestTarget = nil
    local bestScore = -999999
    local maxDist = radius or TP_Click_Radius

    for _, v in ipairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character then
             local hrp = v.Character:FindFirstChild(STR_HRP)
             local hum = v.Character:FindFirstChild(STR_HUM)
             
             if hrp and hum and hum.Health > 0 then
                local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    local screenDist = (Vector2_new(screenPos.X, screenPos.Y) - mouseLoc).Magnitude
                    
                    if screenDist < maxDist then
                        -- Score Calculation:
                        -- Higher is better.
                        -- 1. Closeness to mouse is most important (max 1000 pts)
                        -- 2. Closeness to camera (depth) breaks ties (max 500 pts)
                        
                        local depth = screenPos.Z -- Distance from camera in studs
                        local distScore = (maxDist - screenDist) * 5 -- Weight 5x
                        local depthScore = (1000 - math.clamp(depth, 0, 1000)) * 2 -- Weight 2x (Closer = Higher Score)
                        
                        local totalScore = distScore + depthScore
                        
                        if totalScore > bestScore then
                            bestScore = totalScore
                            bestTarget = v
                        end
                    end
                end
             end
        end
    end
    
    return bestTarget
end

local function StopCoreLoop() if CoreLoopConnection then CoreLoopConnection:Disconnect(); CoreLoopConnection = nil end; FullCleanup() end
local CORE_LOOP_THROTTLE = 0.033
local LastCoreLoopRun = 0

local function StartCoreLoop()
    if CoreLoopConnection then return end 
    
    CoreLoopConnection = RunService.RenderStepped:Connect(function()
        local startTime = os_clock()
        if startTime - LastCoreLoopRun < CORE_LOOP_THROTTLE then return end
        LastCoreLoopRun = startTime
        
        local isAnyModeActive = (HitboxMode or LockMode or EspMode or PlayerEspMode)
        if not isAnyModeActive then StopCoreLoop(); return end

        local myPlayers = Players:GetPlayers()
        for i=1, #myPlayers do
            local v = myPlayers[i]
            if v ~= LocalPlayer and v.Character then
                local hrp = v.Character:FindFirstChild(STR_HRP)
                if hrp then
                    if HitboxMode then
                        local isWL = Whitelist[v.Name]; local shouldExpand = false
                        if not isWL then if HitboxStrategy == "All" then shouldExpand = true elseif HitboxStrategy == "Aim" and v == TargetCache then shouldExpand = true end end
                        
                        if shouldExpand then 
                            SetProp(hrp, "Size", CachedSize)
                            SetProp(hrp, "Transparency", CachedTransparency)
                            SetProp(hrp, "Color", CachedColor)
                            SetProp(hrp, "Material", Enum.Material.Neon)
                            SetProp(hrp, "CanCollide", false)
                        else 
                            SetProp(hrp, "Size", DefaultSize)
                            SetProp(hrp, "Transparency", 1)
                            SetProp(hrp, "Material", Enum.Material.Plastic)
                        end
                    end
                    
                    if EspMode or PlayerEspMode then
                        local col, line, name = ColorWhite, false, false
                        if EspMode then
                            local char = v.Character
                            local isDark = CollectionService:HasTag(char, "DarkholdOwner") or char:FindFirstChild("Darkhold"); local isArt = CollectionService:HasTag(char, "HasArtifact")
                            if isDark then 
                                col=ColorRed; line=true; name=true 
                            elseif isArt then 
                                local abilityFolder = char:FindFirstChild("AbilityFolder")
                                local passive1 = abilityFolder and abilityFolder:GetAttribute("Passive1") or ""
                                if char:GetAttribute("TimeStone/LastUpdate") then col=ColorGreen elseif passive1 == "MeteorRain" then col=ColorBlue else col=ColorPurple end
                                line=true; name=true 
                            end 
                        end
                        if PlayerEspMode and not line then name=true; col=ColorWhite end
                        
                        if line or name then ManageESP(v, col, line, name) else ClearVisuals(v) end
                    else 
                        ClearVisuals(v) 
                    end
                end
            end
        end
        Debug_Metrics.CoreLoopTime = (os_clock() - startTime) * 1000
    end)
    table_insert(_G.SkyL_Connections, CoreLoopConnection)
    
    LogicLoopConnection = task_spawn(function()
        while CoreLoopConnection do
            if LockMode or HitboxMode then
                local target = getClosestPlayerToMouse()
                if LockMode then 
                    if target then SavedTarget = target end; 
                    if SavedTarget and SavedTarget.Character and SavedTarget.Character:FindFirstChild(STR_HUM) and SavedTarget.Character.Humanoid.Health > 0 then 
                        TargetCache = SavedTarget 
                    else 
                        TargetCache = nil; SavedTarget = nil 
                    end
                else TargetCache = target end
            else TargetCache = nil end
            task_wait(0.2) 
        end
    end)
    UpdateConnectionCount()
end

local function SetupSlider(frame, range, default, updateFunc, id)
    local btn = frame:FindFirstChildOfClass("TextButton"); local dragging = false; local fill = frame:FindFirstChild("Fill"); local valLabel = frame:FindFirstChild("Val"); local inputField = frame.Parent:FindFirstChild("InputVal")
    local function UpdateVal(pos, skipCallback) 
        pos = math_clamp(pos, 0, 1)
        fill.Size = UDim2_new(pos, 0, 1, 0)
        local val = math_floor(pos * range)
        valLabel.Text = tostring(val)
        if inputField then inputField.Text = tostring(val) end
        if not skipCallback then updateFunc(val) end
    end
    
    if id then SliderRegistry[id] = {Update = UpdateVal, Range = range} end

    if default == 0 then fill.Size = UDim2_new(0,0,1,0); valLabel.Text = "0"; if inputField then inputField.Text = "0" end end
    btn.InputBegan:Connect(function(input) 
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then 
            dragging = true; LastInteraction = tick(); UpdateVal((input.Position.X - frame.AbsolutePosition.X) / frame.AbsoluteSize.X); 
            local con; con = input.Changed:Connect(function() 
                if input.UserInputState == Enum.UserInputState.End then 
                    dragging = false; 
                    con:Disconnect()
                    SaveSettings() -- Auto Save
                end 
            end) 
        end 
    end)
    UserInputService.InputChanged:Connect(function(input) if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then UpdateVal((input.Position.X - frame.AbsolutePosition.X) / frame.AbsoluteSize.X); LastInteraction = tick() end end)
    if inputField then inputField.FocusLost:Connect(function() local num = tonumber(inputField.Text); if num then num = math_clamp(math_floor(num), 0, range); UpdateVal(num / range); SaveSettings() else inputField.Text = valLabel.Text end end) end
end

        if InvisibleKeyReady then
            ToggleInvisible()
        end
    end

    if io.KeyCode == Enum.KeyCode.H then
        Farm.NPCEnabled = not Farm.NPCEnabled
        if Farm.NPCEnabled then Farm.StartNPC(); AddLog("NPC FARM: ON", ColorGreen) else Farm.StopNPC(); AddLog("NPC FARM: OFF", ColorRed) end
        SyncVisuals()
    end
    if io.KeyCode == Enum.KeyCode.J then
        Farm.ChestEnabled = not Farm.ChestEnabled
        if Farm.ChestEnabled then 
            if not Farm.NPCEnabled then Farm.EnterSafeZone() end
            AddLog("CHEST FARM: ON", ColorGreen) 
        else 
            Farm.ExitSafeZone()
            AddLog("CHEST FARM: OFF", ColorRed) 
        end
        SyncVisuals()
    end
    
    if TeleportMode then
        if (io.UserInputType == Enum.UserInputType.Touch or io.UserInputType == Enum.UserInputType.MouseButton1) then
            local now = tick()
            local shouldTP = false
            
            if TP_PC_Mode then
                shouldTP = true
            elseif TP_Mobile_Mode then
                local currentPos = Vector2_new(io.Position.X, io.Position.Y)
                local dist = (currentPos - LastTPPos).Magnitude

                if (now - LastTPClick) < TP_DoubleTapSpeed and dist < TP_Tap_Threshold then 
                    shouldTP = true
                    LastTPClick = 0 
                else 
                    LastTPClick = now
                    LastTPPos = currentPos
                end
            elseif TP_Aim_Mode then
                 shouldTP = true
            else
               if (now - LastTPClick) < TP_DoubleTapSpeed then shouldTP = true; LastTPClick = 0 else LastTPClick = now end
            end

            if shouldTP then
                local targetToTp = nil
                
                if TP_Aim_Mode then
                    targetToTp = getClosestPlayerToCenter()
                else
                    -- [STICKY] Use Focus Target if valid, otherwise scan
                    if TP_Current_Focus and TP_Current_Focus.Parent then
                         targetToTp = TP_Current_Focus
                    else
                         local clickPos = io.Position; local mousePos = Vector2_new(clickPos.X, clickPos.Y)
                         targetToTp = GetSmartTarget(mousePos, TP_Click_Radius)
                    end
                end
                
                if targetToTp and targetToTp.Character and targetToTp.Character:FindFirstChild(STR_HRP) and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(STR_HRP) then 
                    local destCF = targetToTp.Character.HumanoidRootPart.CFrame * CFrame_new(0,0,3)
                    if InvisibleMode then
                        InvisibleDesiredCFrame = destCF
                        AddLog("uir!", ColorGreen)
                    else
                        LocalPlayer.Character.HumanoidRootPart.CFrame = destCF
                        AddLog("uir!", ColorGreen)
                    end
                end
            end
        end
    end
end))

-- // UPDATE FOCUS VISUALS OTIMIZADO // --
local function UpdateFocusVisuals()
    if not (TeleportMode and (EspMode or PlayerEspMode)) then return end

    local focusTarget = nil

    if TP_Aim_Mode then
        focusTarget = getClosestPlayerToCenter()
    else
        focusTarget = GetSmartTarget(nil, TP_Click_Radius)
    end
    
    -- [STICKY] Update Global Focus
    TP_Current_Focus = focusTarget

    if focusTarget and focusTarget.Character then
        local head = focusTarget.Character:FindFirstChild(STR_HEAD)
        local hrp = focusTarget.Character:FindFirstChild(STR_HRP)
        
        if head and head:FindFirstChild("ArtifactNameTag") then
             local tag = head.ArtifactNameTag:FindFirstChild("L")
             if tag then SetProp(tag, "TextColor3", ColorYellow) end
        end

        if hrp and hrp:FindFirstChild("ArtifactBeam") then
             SetProp(hrp.ArtifactBeam, "Color", ColorSequence_new(ColorYellow))
        end
    end
end

    if now - lastFocusUpdate < 0.15 then return end
    lastFocusUpdate = now
    UpdateFocusVisuals()
end))
